data-rq-each

Render a list from an array in state. Put a <template> in your HTML, give it data-rq-each, and reQuery will stamp out one copy of the template for every item in the array. When the array changes, the list re-renders automatically.

<template data-rq-each="key">
  …item template…
</template>

What's available inside the template

Array item type Keys you can bind to inside the template
Plain values ("hello", 42) value — the item itself
$index — the item's position (0-based)
Objects ({ name: "Alice" }) Any key on the object directly (e.g. name, email)
$index — position
1

Simple text list

html 
<div id="app">
  <input id="item-input" type="text" placeholder="Add a fruit…" />
  <button id="add-btn">Add</button>

  <ul>
    <template data-rq-each="fruits">
      <li><span data-rq-text="value"></span></li>
    </template>
  </ul>
</div>
js 
$('#app').rqState({ fruits: ['Apple', 'Banana', 'Cherry'] });

$('#add-btn').on('click', function () {
  var input = $('#item-input');
  var text = input.val().trim();
  if (!text) return;

  $('#app').rqMutate('fruits', function (list) {
    return [...list, text];
  });

  input.val('');
});

Object array — team roster

html 
<div id="app">
  <table>
    <thead>
      <tr><th>Name</th><th>Role</th></tr>
    </thead>
    <tbody>
      <template data-rq-each="team">
        <tr>
          <td data-rq-text="name"></td>
          <td data-rq-text="role"></td>
        </tr>
      </template>
    </tbody>
  </table>

  <button id="add-member">Add member</button>
</div>
js 
$('#app').rqState({
  team: [
    { name: 'Alice', role: 'Designer' },
    { name: 'Bob',   role: 'Developer' }
  ]
});

var newMembers = [
  { name: 'Carol', role: 'Manager' },
  { name: 'Dave',  role: 'QA' }
];
var i = 0;

$('#add-member').on('click', function () {
  if (i >= newMembers.length) return;
  $('#app').rqMutate('team', function (list) {
    return [...list, newMembers[i++]];
  });
});
